# -*- coding: utf-8 -*-
from odoo import models, fields, api
from datetime import datetime, timedelta


class RentalForecast(models.Model):
    _name = 'rental.forecast'
    _description = 'Prévision de Revenus'
    _order = 'month'

    month = fields.Date("Mois", required=True)

    # Données réelles (après coup)
    actual_occupancy = fields.Float("Occupation Réelle (%)")
    actual_revenue = fields.Float("Revenu Réel (DA)")

    # Prévisions
    predicted_occupancy = fields.Float("Occupation Prédite (%)")
    predicted_revenue = fields.Float("Revenu Prédit (DA)")

    # Variance
    variance = fields.Float("Écart (%)", compute="_compute_variance")
    variance_amount = fields.Float("Écart (DA)", compute="_compute_variance")

    # Analyse
    trend = fields.Selection([
        ('up', '📈 Hausse'),
        ('stable', '➡️ Stable'),
        ('down', '📉 Baisse'),
    ], string="Tendance", compute="_compute_trend")

    @api.depends('actual_revenue', 'predicted_revenue')
    def _compute_variance(self):
        for rec in self:
            if rec.predicted_revenue > 0:
                rec.variance = ((rec.actual_revenue - rec.predicted_revenue) / rec.predicted_revenue) * 100
                rec.variance_amount = rec.actual_revenue - rec.predicted_revenue
            else:
                rec.variance = 0
                rec.variance_amount = 0

    @api.depends('predicted_revenue')
    def _compute_trend(self):
        for rec in self:
            # Comparer avec le mois précédent
            previous = self.search([
                ('month', '<', rec.month),
            ], order='month desc', limit=1)

            if previous:
                if rec.predicted_revenue > previous.predicted_revenue * 1.05:
                    rec.trend = 'up'
                elif rec.predicted_revenue < previous.predicted_revenue * 0.95:
                    rec.trend = 'down'
                else:
                    rec.trend = 'stable'
            else:
                rec.trend = 'stable'

    @api.model
    def generate_forecast(self, months_ahead=6):
        """Génère les prévisions pour les mois à venir"""
        today = fields.Date.today()

        for i in range(months_ahead):
            month_date = today.replace(day=1) + timedelta(days=32 * i)
            month_date = month_date.replace(day=1)

            # Analyser l'historique des 3 dernières années pour ce mois
            historical = self.search([
                ('month', '>=', month_date.replace(year=month_date.year - 3)),
                ('month', '<', month_date),
                ('actual_revenue', '>', 0),
            ])

            if historical:
                avg_revenue = sum(historical.mapped('actual_revenue')) / len(historical)
                avg_occupancy = sum(historical.mapped('actual_occupancy')) / len(historical)
            else:
                avg_revenue = 0
                avg_occupancy = 0

            # Créer ou mettre à jour la prévision
            forecast = self.search([('month', '=', month_date)], limit=1)
            vals = {
                'predicted_revenue': avg_revenue,
                'predicted_occupancy': avg_occupancy,
            }

            if forecast:
                forecast.write(vals)
            else:
                vals['month'] = month_date
                self.create(vals)

        return True---------- END ----------

        """Retourne le prix ajusté pour une date donnée (saisonnalité)"""
        self.ensure_one()
        seasonal = self.seasonal_rate_ids.filtered(
            lambda r: r.is_active and r.date_start <= date <= r.date_end
        )
        if seasonal:
            adjustment = seasonal[0].price_adjustment / 100.0
            return self.price_month * (1 + adjustment)
        return self.price_month

    def get_best_price(self, date_start, date_end):
        """Calcule le meilleur prix pour une période donnée"""
        self.ensure_one()
        days = (date_end - date_start).days
        if days <= 0:
            return 0, 'day'

        # Prix de base
        prices = {
            'day': self.price_day,
            'week': self.price_week,
            'month': self.price_month,
            'year': self.price_year,
        }

        # Appliquer saisonnalité
        seasonal_adjustment = 0
        seasonal = self.seasonal_rate_ids.filtered(
            lambda r: r.is_active and r.date_start <= date_start <= r.date_end
        )
        if seasonal:
            seasonal_adjustment = seasonal[0].price_adjustment / 100.0

        # Calculer le coût pour chaque période
        costs = {}
        if prices['day'] > 0:
            costs['day'] = days * prices['day']
        if prices['week'] > 0 and days >= 7:
            costs['week'] = (days / 7.0) * prices['week']
        if prices['month'] > 0 and days >= 30:
            costs['month'] = (days / 30.0) * prices['month']
        if prices['year'] > 0 and days >= 365:
            costs['year'] = (days / 365.0) * prices['year']

        if not costs:
            return 0, 'day'

        # Choisir le moins cher
        best_period = min(costs, key=costs.get)
        best_price = costs[best_period] * (1 + seasonal_adjustment)

        return best_price, best_period---------- END ----------

    @api.depends()
    def _compute_kpi(self):
        for rec in self:
            apartments = self.env['property.apartment'].search([])
            rec.total_apartments = len(apartments)
            rec.available_count = len(apartments.filtered(lambda a: a.state == 'available'))
            rec.occupied_count = len(apartments.filtered(lambda a: a.state == 'occupied'))
            rec.soon_count = len(apartments.filtered(lambda a: a.state == 'soon'))
            rec.maintenance_count = len(apartments.filtered(lambda a: a.state == 'maintenance'))
            rec.cleaning_count = len(apartments.filtered(lambda a: a.state == 'cleaning'))

            total = rec.total_apartments or 1
            rec.occupancy_rate = round((rec.occupied_count / total) * 100, 2)

    @api.depends()
    def _compute_revenue(self):
        today = fields.Date.today()
        week_start = today - timedelta(days=today.weekday())
        month_start = today.replace(day=1)
        year_start = today.replace(month=1, day=1)

        for rec in self:
            rentals = self.env['property.rental'].search([('state', '=', 'done')])

            rec.daily_revenue = sum(rentals.filtered(lambda r: r.date_end == today).mapped('total_price'))
            rec.weekly_revenue = sum(rentals.filtered(lambda r: r.date_end >= week_start).mapped('total_price'))
            rec.monthly_revenue = sum(rentals.filtered(lambda r: r.date_end >= month_start).mapped('total_price'))
            rec.yearly_revenue = sum(rentals.filtered(lambda r: r.date_end >= year_start).mapped('total_price'))

                delta = (rec.date_end - rec.date_start).days
                nb_days = max(delta, 0)

                rec.duration_days = nb_days

                if nb_days <= 0:

                    rec.unit_price = 0
                    rec.total_price = 0
                    return

                apt = rec.property_id

                # Utiliser la méthode intelligente de l'appartement
                best_price, best_period = apt.get_best_price(rec.date_start, rec.date_end)

                rec.period_type = best_period
                rec.total_price = best_price

                # Prix unitaire selon période
                period_prices = {
                    'day': apt.price_day,
                    'week': apt.price_week,
                    'month': apt.price_month,
                    'year': apt.price_year,

            if rec.client_id and rec.client_id.rental_count > 3:

    @api.depends('rental_ids', 'rental_ids.state')
    def _compute_rental_stats(self):
        for rec in self:
            rec.rental_count = len(rec.rental_ids)
            rec.active_rentals = len(rec.rental_ids.filtered(
                lambda r: r.state in ('confirmed', 'ongoing', 'expiring')
            ))
            rec.total_spent = sum(rec.rental_ids.filtered(
                lambda r: r.state == 'done'
            ).mapped('total_price'))

            durations = rec.rental_ids.mapped('duration_days')
            rec.avg_rental_duration = sum(durations) / len(durations) if durations else 0

    @api.depends('rental_count', 'total_spent')
    def _compute_loyalty(self):
        for rec in self:
            points = rec.rental_count * 10 + int(rec.total_spent / 1000)
            rec.loyalty_points = points

            if points >= 500:
                rec.loyalty_level = 'platinum'
            elif points >= 200:
                rec.loyalty_level = 'gold'
            elif points >= 50:
                rec.loyalty_level = 'silver'
            else:
                rec.loyalty_level = 'bronze'

    @api.depends('rental_ids', 'rental_ids.payment_status')
    def _compute_risk(self):
        for rec in self:
            if not rec.rental_ids:
                rec.risk_score = 0
                rec.risk_level = 'low'
                continue

            # Calculer le score de risque
            score = 0
            total = len(rec.rental_ids)

            # Retards de paiement
            late_payments = len(rec.rental_ids.filtered(lambda r: r.payment_status == 'unpaid'))
            score += (late_payments / total) * 50 if total else 0

            # Locations annulées
            cancelled = len(rec.rental_ids.filtered(lambda r: r.state == 'cancelled'))
            score += (cancelled / total) * 30 if total else 0

            # Dégâts
            damages = len(rec.rental_ids.filtered(lambda r: r.inspection_ids.filtered(lambda i: i.damage_found)))
            score += (damages / total) * 20 if total else 0

            rec.risk_score = min(int(score), 100)

            if score < 20:
                rec.risk_level = 'low'
            elif score < 50:
                rec.risk_level = 'medium'
            else:
                rec.risk_level = 'high'---------- END ----------

# -*- coding: utf-8 -*-
from odoo import http
from odoo.http import request


class RentalPublicForm(http.Controller):

    @http.route('/rental/form', type='http', auth='public', website=True)
    def rental_form(self, **kwargs):
        properties = request.env['property.apartment'].sudo().search([
            ('state', '=', 'available')
        ])
        return request.render('rental_management.public_rental_form', {
            'properties': properties,
        })

    @http.route('/rental/submit', type='http', auth='public', website=True, methods=['POST'])
    def rental_submit(self, **post):
        error = None
        success = None

        try:
            # Créer le client s'il n'existe pas
            partner = request.env['res.partner'].sudo().create({
                'name': post.get('name'),
                'email': post.get('email'),
                'phone': post.get('phone'),
                'is_tenant': True,
            })

            # Créer la demande de location (en brouillon)
            rental = request.env['property.rental'].sudo().create({
                'property_id': int(post.get('property_id')),
                'client_id': partner.id,
                'date_start': post.get('start_date'),
                'date_end': post.get('end_date'),
                'notes': post.get('message', ''),
            })

            success = "Votre demande a été enregistrée ! N° de contrat: %s" % rental.name

        except Exception as e:
            error = str(e)

        properties = request.env['property.apartment'].sudo().search([
            ('state', '=', 'available')
        ])

        return request.render('rental_management.public_rental_form', {
            'properties': properties,
            'success': success,
            'error': error,
        })---------- END ----------
